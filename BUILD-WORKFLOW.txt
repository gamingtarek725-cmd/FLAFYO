name: build
on: [push, pull_request, workflow_dispatch]
jobs:
  windows:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-dotnet@v4
        with: { dotnet-version: '8.0.x' }
      - name: Run tests
        run: dotnet test tests/Flafyo.Core.Tests -c Release
      - name: Build EXE
        run: dotnet publish src/Flafyo.App/Flafyo.App.csproj -c Release -r win-x64 --self-contained true -p:PublishSingleFile=true -p:IncludeNativeLibrariesForSelfExtract=true -o publish
      - name: SHA256
        run: Get-FileHash publish/FLAFYO.GameTranslator.exe -Algorithm SHA256 | Format-List
      - uses: actions/upload-artifact@v4
        with:
          name: FLAFYO-GAME-TRANSLATOR
          path: publish/FLAFYO.GameTranslator.exe
      - name: Build installer (Setup.exe)
        continue-on-error: true
        shell: pwsh
        run: |
          choco install innosetup -y --no-progress
          & "${env:ProgramFiles(x86)}\Inno Setup 6\ISCC.exe" installer\FLAFYO.iss
      - uses: actions/upload-artifact@v4
        continue-on-error: true
        with:
          name: FLAFYO-Setup
          path: installer-out/FLAFYO-GAME-TRANSLATOR-Setup.exe
      - name: Download offline language pack (for the real-model test)
        continue-on-error: true
        shell: pwsh
        run: |
          $base = "https://huggingface.co/Xenova/opus-mt-en-ar/resolve/63765349ac9f6c43f6e5bd8c0a5583224cfd8587"
          $dir = Join-Path $env:RUNNER_TEMP "model"
          New-Item -ItemType Directory -Force (Join-Path $dir "onnx") | Out-Null
          foreach ($f in "config.json","generation_config.json","source.spm","target.spm","vocab.json","onnx/encoder_model_quantized.onnx","onnx/decoder_model_quantized.onnx") {
            Invoke-WebRequest "$base/$f" -OutFile (Join-Path $dir $f)
          }
          "FLAFYO_MODEL_DIR=$dir" | Out-File -FilePath $env:GITHUB_ENV -Append -Encoding utf8
      - name: Real offline translation test
        continue-on-error: true
        run: dotnet test tests/Flafyo.Core.Tests -c Release --filter "FullyQualifiedName~Integration" --logger "console;verbosity=detailed"
